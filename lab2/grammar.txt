Program         -> (Function | Formula)*

Function        -> 'function' ID '(' VarDefs ')' ['requires' Predicate] 'returns' VarDefs ['ensures' Predicate] ['uses' VarDefs] Statement
                 | ID '(' VarDefs ')' ['requires' Predicate] 'returns' VarDefs ['ensures' Predicate] ['uses' VarDefs] Statement

Formula         -> ID '(' VarDefs ')' '=>' Predicate ';'

VarDefs         -> [VarDef (',' VarDef)*]
VarDef          -> ID ':' Type
Type            -> ID ['[' ']']

Statement       -> Block | IfStmt | WhileStmt | AssertStmt | AssumeStmt | Assignment

Block           -> '{' Statement* '}'

IfStmt          -> 'if' '(' Condition ')' Statement ['else' Statement]

WhileStmt       -> 'while' '(' Condition ')' ['invariant' Predicate] Statement

AssertStmt      -> 'assert' Expr ';'

AssumeStmt      -> 'assume' Expr ';'

Assignment      -> TupleAssign | ArrayAssign | SimpleAssign

TupleAssign     -> ID (',' ID)* '=' FuncCall ';'
ArrayAssign     -> ID '[' Expr ']' '=' Expr ';'
SimpleAssign    -> ID '=' Expr ';'

Condition       -> OrExpr ['->' Condition]
OrExpr          -> AndExpr ('or' AndExpr)*
AndExpr         -> NotExpr ('and' NotExpr)*
NotExpr         -> 'not' NotExpr | Comparison
Comparison      -> Expr [('==' | '!=' | '>=' | '<=' | '>' | '<') Expr]

Expr            -> Additive
Additive        -> Multiplicative (('+' | '-') Multiplicative)*
Multiplicative  -> Unary (('*' | '/') Unary)*
Unary           -> '-' Unary | Postfix

Postfix         -> Primary ('[' Expr ']')*

Primary         -> INT 
                 | TRUE 
                 | FALSE 
                 | 'length' '(' Expr ')' 
                 | ID 
                 | '(' Expr ')' 
                 | FuncCall

FuncCall        -> ID '(' ExprList ')'
ExprList        -> [Expr (',' Expr)*]

Predicate       -> PredOr ['->' Predicate]
PredOr          -> PredAnd ('or' PredAnd)*
PredAnd         -> PredNot ('and' PredNot)*
PredNot         -> 'not' PredNot | PredAtom

PredAtom        -> TRUE 
                 | FALSE 
                 | Quantifier 
                 | '(' Predicate ')' 
                 | FormulaRef 
                 | ComparisonInPred

FormulaRef      -> ID '(' ExprList ')'
ComparisonInPred-> Expr [('==' | '!=' | '>=' | '<=' | '>' | '<') Expr]

Quantifier      -> ('forall' | 'exists') '(' ID ':' Type Predicate ')'
